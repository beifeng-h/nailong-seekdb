cmake_minimum_required(VERSION 3.20)
project(minivectordb LANGUAGES CXX)

add_library(minivectordb src/page_store.cpp src/database.cpp src/sql_parser.cpp src/sql_executor.cpp)
target_include_directories(minivectordb PUBLIC include)
target_compile_features(minivectordb PUBLIC cxx_std_20)
if(MSVC)
  target_compile_options(minivectordb PRIVATE /W4 /WX)
  target_compile_options(minivectordb PUBLIC /utf-8)
else()
  target_compile_options(minivectordb PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

enable_testing()
add_executable(storage_tests tests/storage_tests.cpp)
target_link_libraries(storage_tests PRIVATE minivectordb)
add_test(NAME storage_contracts COMMAND storage_tests)
add_executable(storage_fault_driver tests/storage_fault_driver.cpp)
target_link_libraries(storage_fault_driver PRIVATE minivectordb)
add_executable(database_tests tests/database_tests.cpp)
target_link_libraries(database_tests PRIVATE minivectordb)
add_test(NAME database_contracts COMMAND database_tests)
add_executable(database_fault_driver tests/database_fault_driver.cpp)
target_link_libraries(database_fault_driver PRIVATE minivectordb)
add_executable(sql_tests tests/sql_tests.cpp)
target_link_libraries(sql_tests PRIVATE minivectordb)
add_test(NAME sql_contracts COMMAND sql_tests)
add_executable(sql_fault_driver tests/sql_fault_driver.cpp)
target_link_libraries(sql_fault_driver PRIVATE minivectordb)
add_executable(sql_reference_tests tests/sql_reference_tests.cpp)
target_link_libraries(sql_reference_tests PRIVATE minivectordb)
add_test(NAME sql_reference_contracts COMMAND sql_reference_tests)
add_executable(sql_public_case_driver tests/sql_public_case_driver.cpp)
target_link_libraries(sql_public_case_driver PRIVATE minivectordb)

# Keep internal definitions in the test's primary translation unit for GCC.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS src/page_store.cpp tests/file_io_tests.cpp)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/page_store.cpp" file_io_implementation)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/tests/file_io_tests.cpp" file_io_source)
string(REPLACE "#include \"../src/page_store.cpp\"" "${file_io_implementation}" file_io_source "${file_io_source}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/file_io_tests.cpp" "${file_io_source}")
add_executable(file_io_tests "${CMAKE_CURRENT_BINARY_DIR}/file_io_tests.cpp")
target_include_directories(file_io_tests PRIVATE include)
target_compile_features(file_io_tests PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(file_io_tests PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(file_io_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME file_io_contracts COMMAND file_io_tests "${CMAKE_CURRENT_BINARY_DIR}/file-io-test-data")
